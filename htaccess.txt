# ============================================================
#  atma.marketing — New Market Physics
#  Apache config for low-latency static serving (GoDaddy/cPanel)
# ============================================================

# ---- 1. Compression (biggest latency win: ~840KB -> ~190KB) ----
<IfModule mod_brotli.c>
  AddOutputFilterByType BROTLI_COMPRESS text/html text/css application/javascript application/json image/svg+xml
</IfModule>
<IfModule mod_deflate.c>
  AddOutputFilterByType DEFLATE text/html text/css application/javascript application/json image/svg+xml
</IfModule>

# ---- 2. Caching ----
<IfModule mod_expires.c>
  ExpiresActive On
  # HTML: revalidate every load so edits go live immediately
  ExpiresByType text/html "access plus 0 seconds"
  # Vendored React (version-pinned) + any future split CSS/JS
  ExpiresByType text/css "access plus 1 hour"
  ExpiresByType application/javascript "access plus 1 hour"
  # Static binaries: cache hard (rename the file to bust the cache)
  ExpiresByType image/png  "access plus 1 year"
  ExpiresByType image/jpeg "access plus 1 year"
  ExpiresByType image/webp "access plus 1 year"
  ExpiresByType font/woff2 "access plus 1 year"
</IfModule>

<IfModule mod_headers.c>
  <FilesMatch "\.(png|jpe?g|webp|woff2?)$">
    Header set Cache-Control "public, max-age=31536000, immutable"
  </FilesMatch>
  <FilesMatch "^react(-dom)?\.production\.min\.js$">
    Header set Cache-Control "public, max-age=31536000, immutable"
  </FilesMatch>
  <FilesMatch "\.html$">
    Header set Cache-Control "no-cache, must-revalidate"
  </FilesMatch>
  # Light hardening
  Header always set X-Content-Type-Options "nosniff"
  Header always set Referrer-Policy "strict-origin-when-cross-origin"
</IfModule>

# ---- 3. Defaults ----
DirectoryIndex index.html
AddDefaultCharset UTF-8
